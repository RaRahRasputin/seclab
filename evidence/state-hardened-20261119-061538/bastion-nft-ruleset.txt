table inet filter {
	counter in_drop {
		packets 0 bytes 0
	}

	counter out_drop {
		packets 2 bytes 106
	}

	chain input {
		type filter hook input priority filter; policy drop;
		iifname "lo" accept
		ct state established,related accept
		ct state invalid drop
		ip saddr 10.10.200.0/24 tcp dport 22 accept
		ip saddr 10.10.99.20 tcp dport 9100 accept
		icmp type echo-request limit rate 5/second burst 5 packets accept
		counter name "in_drop" drop
	}

	chain output {
		type filter hook output priority filter; policy drop;
		oifname "lo" accept
		ct state established,related accept
		ip daddr 10.10.0.0/16 tcp dport 22 accept
		ip daddr 10.10.99.20 tcp dport { 3000, 9090 } accept
		counter name "out_drop" drop
	}

	chain forward {
		type filter hook forward priority filter; policy drop;
	}
}
table inet f2b-table {
	set addr-set-sshd {
		type ipv4_addr
	}

	chain f2b-chain {
		type filter hook input priority filter - 1; policy accept;
		tcp dport 22 ip saddr @addr-set-sshd reject with icmp port-unreachable
	}
}
